                let cred = await createUserWithEmailAndPassword(auth, email, pass);
                let uid = cred.user.uid;
                await set(ref(db, 'usernames/' + uname.toLowerCase()), uid);
                await set(ref(db, 'users/' + uid), { username: uname, email: email, profileLikes: 0, avatar: 'https://via.placeholder.com/70' });
                alert('সাইন আপ সফল হয়েছে! 🎉');
            } catch(e) { alert('ত্রুটি: ' + e.message); }
        }

        window.loginUser = async () => {
            try { await signInWithEmailAndPassword(auth, document.getElementById('loginEmail').value, document.getElementById('loginPassword').value); } 
            catch(e) { alert('লগইন ব্যর্থ: ' + e.message); }
        }

        window.logoutUser = async () => { await signOut(auth); closeSettingsModal(); }

        onAuthStateChanged(auth, (user) => {
            if(user) {
                currentUser = user;
                document.getElementById('authPage').classList.remove('active');
                document.getElementById('bottomNav').style.display = 'flex';
                switchPage('homePage');
            } else {
                currentUser = null;
                document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
                document.getElementById('authPage').classList.add('active');
                document.getElementById('bottomNav').style.display = 'none';
            }
        });

        window.uploadReel = async () => {
            let vUrl = document.getElementById('videoUrl').value, title = document.getElementById('videoTitle').value, buyUrl = document.getElementById('buyNowUrl').value;
            if(!vUrl || !title) { alert('ভিডিও লিংক ও টাইটেল দিন!'); return; }
            let userSnap = await get(ref(db, 'users/' + currentUser.uid));
            let uData = userSnap.val();
            let newRef = push(ref(db, 'reels'));
            await set(newRef, { reelId: newRef.key, uid: currentUser.uid, username: uData.username, avatar: uData.avatar || 'https://via.placeholder.com/40', videoUrl: vUrl, title: title, buyNowUrl: buyUrl || '#', likes: 0, dislikes: 0 });
            alert('রিলস আপলোড সফল! 🚀');
            document.getElementById('videoUrl').value = ''; document.getElementById('videoTitle').value = ''; document.getElementById('buyNowUrl').value = '';
            switchPage('homePage');
        }

        window.loadReelsFeed = async () => {
            let feed = document.getElementById('reelsFeed');
            feed.innerHTML = '<p style="text-align:center; margin-top:50px;">লোড হচ্ছে...</p>';
            let snap = await get(ref(db, 'reels'));
            if(!snap.exists()) { feed.innerHTML = '<p style="text-align:center; margin-top:50px;">কোনো রিলস নেই!</p>'; return; }
            
            feed.innerHTML = '';
            snap.forEach((child) => {
                let r = child.val();
                let card = document.createElement('div');
                card.className = 'reel-card';
                card.innerHTML = `
                    <video src="${r.videoUrl}" loop autoplay muted playsinline onclick="this.paused ? this.play() : this.pause()"></video>
                    <div class="reel-info">
                        <div class="creator-info" onclick="viewProfile('${r.uid}')">
                            <div class="creator-avatar"><img src="${r.avatar}" alt=""></div>
                            <b>@${r.username}</b>
                        </div>
                        <p style="font-size: 14px; margin-bottom: 5px;">${r.title}</p>
                        <button class="buy-now-btn" onclick="window.open('${r.buyNowUrl}', '_blank')">Buy Now</button>
                    </div>
                    <div class="action-buttons">
                        <button class="action-btn" onclick="reactReel('${r.reelId}', 'like')">👍<span>${r.likes || 0}</span></button>
                        <button class="action-btn" onclick="reactReel('${r.reelId}', 'dislike')">👎<span>${r.dislikes || 0}</span></button>
                    </div>
                `;
                feed.appendChild(card);
            });
        }

        window.reactReel = async (reelId, type) => {
            let rRef = ref(db, 'reels/' + reelId);
            let snap = await get(rRef);
            if(snap.exists()) {
                let d = snap.val();
                let likes = d.likes || 0, dislikes = d.dislikes || 0;
                if(type === 'like') likes++; else dislikes++;
                await update(rRef, { likes, dislikes });
                loadReelsFeed();
            }
        }

        window.viewProfile = async (uid) => {
            viewingProfileUid = uid;
            document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
            document.getElementById('profilePage').classList.add('active');
            document.querySelectorAll('.nav-item').forEach(n => n.classList.remove('active'));
            document.querySelectorAll('.nav-item')[2].classList.add('active');

            let userSnap = await get(ref(db, 'users/' + uid));
            if(!userSnap.exists()) return;
            let u = userSnap.val();
            document.getElementById('profileUsernameDisplay').innerText = '@' + u.username;
            document.getElementById('profileEmailDisplay').innerText = u.email;
            document.getElementById('profileLikeCount').innerText = u.profileLikes || 0;
            document.getElementById('profileImg').src = u.avatar || 'https://via.placeholder.com/70';

            let likeBtn = document.getElementById('profileLikeBtnAction');
            likeBtn.style.display = (uid === currentUser.uid) ? 'none' : 'block';

            let grid = document.getElementById('userGridVideos');
            grid.innerHTML = '';
            let reelsSnap = await get(ref(db, 'reels'));
            let count = 0;
            if(reelsSnap.exists()) {
                reelsSnap.forEach((child) => {
                    let r = child.val();
                    if(r.uid === uid) {
                        count++;
                        let item = document.createElement('div');
                        item.className = 'grid-video-card';
                        item.innerHTML = `<video src="${r.videoUrl}"></video>
                        <div style="position:absolute; bottom:5px; left:5px; font-size:11px; background:rgba(0,0,0,0.6); padding:2px 5px; border-radius:3px;">${r.title}</div>`;
                        if(uid === currentUser.uid) {
                            item.ondblclick = async () => {
                                if(confirm('এই ভিডিওটি ডিলিট করতে চান?')) {
                                    await remove(ref(db, 'reels/' + r.reelId));
                                    viewProfile(uid);
                                }
                            }
                        }
                        grid.appendChild(item);
                    }
                });
            }
            document.getElementById('totalVideosCount').innerText = count;
        }

        window.toggleProfileLike = async () => {
            let uRef = ref(db, 'users/' + viewingProfileUid);
            let snap = await get(uRef);
            if(snap.exists()) {
                let currentLikes = snap.val().profileLikes || 0;
                await update(uRef, { profileLikes: currentLikes + 1 });
                viewProfile(viewingProfileUid);
            }
        }

        window.openSettingsModal = () => { document.getElementById('settingsModal').style.display = 'flex'; }
        window.closeSettingsModal = () => { document.getElementById('settingsModal').style.display = 'none'; }
        
        window.saveSettings = async () => {
            let newAvatar = document.getElementById('editAvatarUrl').value;
            let updates = {};
            if(newAvatar) updates.avatar = newAvatar;
            if(Object.keys(updates).length > 0) {
                await update(ref(db, 'users/' + currentUser.uid), updates);
                alert('প্রোফাইল আপডেট সফল!');
                closeSettingsModal();
                viewProfile(currentUser.uid);
            }
        }
